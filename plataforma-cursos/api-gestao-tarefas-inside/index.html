API de Gestão de Tarefas - Backend

Stack: Node.js + Express + MongoDB (NoSQL) + API REST

// server.js - 60 linhas - Arquitetura MVC
const express = require('express');
const mongoose = require('mongoose');
const app = express();
app.use(express.json());

const TarefaSchema = new mongoose.Schema({
  titulo: String, status: String
});
const Tarefa = mongoose.model('Tarefa', TarefaSchema);

app.get('/tarefas', async (req,res)=>{
  const dados = await Tarefa.find(); // NoSQL
  res.json(dados);
});
app.post('/tarefas', async (req,res)=>{
  const nova = await Tarefa.create(req.body);
  res.status(201).json(nova);
});
// SQL exemplo: SELECT status, COUNT(*) FROM tarefas GROUP BY status
app.listen(3000);

Endpoints: GET /tarefas, POST /tarefas, PUT /tarefas/:id, DELETE /tarefas/:id
